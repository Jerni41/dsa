import random
import math
from typing import Tuple, List

class VigenereCipher:
    
    def __init__(self):
        self.alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'
        self.alphabet_lower = 'abcdefghijklmnopqrstuvwxyz'
    
    def prepare_key(self, text: str, key: str) -> str:
        key = key.upper()
        prepared_key = ""
        key_index = 0
        
        for char in text:
            if char.isalpha():
                # Берем по индексу с циклическим повторением (остаток от деления)
                prepared_key += key[key_index % len(key)]
                key_index += 1
            else:
                # Если символ не буква (пробел, знак), оставляем его без изменений
                prepared_key += char
                
        return prepared_key
    
    def encrypt(self, plaintext: str, key: str) -> str:
        """
        Шифрует открытый текст с использованием ключа
        plaintext: исходный текст для шифрования
        key: ключ шифрования
        Возвращает зашифрованный текст
        """
        prepared_key = self.prepare_key(plaintext, key)
        ciphertext = ""
        
        for i, char in enumerate(plaintext):
            if char.isupper():
                shift = self.alphabet.index(prepared_key[i])
                original_pos = self.alphabet.index(char)
                new_pos = (original_pos + shift) % 26
                ciphertext += self.alphabet[new_pos]
            elif char.islower():
                shift = self.alphabet.index(prepared_key[i].upper())
                original_pos = self.alphabet_lower.index(char)
                new_pos = (original_pos + shift) % 26
                ciphertext += self.alphabet_lower[new_pos]
            else:
                ciphertext += char
                
        return ciphertext
    
    def decrypt(self, ciphertext: str, key: str) -> str:
        """
        Расшифровывает зашифрованный текст с использованием ключа
        ciphertext: зашифрованный текст
        key: ключ шифрования
        Возвращает расшифрованный открытый текст
        """
        prepared_key = self.prepare_key(ciphertext, key)
        plaintext = ""
        
        for i, char in enumerate(ciphertext):
            if char.isupper():
                shift = self.alphabet.index(prepared_key[i])
                original_pos = self.alphabet.index(char)
                new_pos = (original_pos - shift) % 26
                plaintext += self.alphabet[new_pos]
            elif char.islower():
                shift = self.alphabet.index(prepared_key[i].upper())
                original_pos = self.alphabet_lower.index(char)
                new_pos = (original_pos - shift) % 26
                plaintext += self.alphabet_lower[new_pos]
            else:
                plaintext += char
                
        return plaintext

### Криптосистема Пэйе

```python
class PaillierCryptosystem:
    
    def __init__(self, key_length: int = 64):
        self.key_length = key_length
    
    def gcd(self, a: int, b: int) -> int:
        """
        Находит наибольший общий делитель (алгоритм Евклида)
        a: первое число
        b: второе число
        Возвращает НОД(a, b)
        """
        while b:
            a, b = b, a % b
        return a
    
    def lcm(self, a: int, b: int) -> int:
        """
        Находит наименьшее общее кратное
        a: первое число
        b: второе число
        Возвращает НОК(a, b)
        """
        return abs(a * b) // self.gcd(a, b)
    
    def mod_inverse(self, a: int, m: int) -> int:
        """
        Находит обратное число по модулю m (такое x, что a*x ≡ 1 mod m)
        a: число, для которого ищем обратное
        m: модуль
        Возвращает обратное число по модулю
        """
        def extended_gcd(a, b):
            """
            Расширенный алгоритм Евклида
            a, b: числа
            Возвращает кортеж (gcd, x, y), где a*x + b*y = gcd
            """
            if a == 0:
                return b, 0, 1
            gcd, x1, y1 = extended_gcd(b % a, a)
            x = y1 - (b // a) * x1
            y = x1
            return gcd, x, y
        
        gcd, x, _ = extended_gcd(a % m, m)
        if gcd != 1:
            raise ValueError("Inverse doesn't exist")
        return (x % m + m) % m
    
    def is_prime(self, n: int, k: int = 5) -> bool:
        """
        Проверяет, является ли число простым с использованием теста Миллера-Рабина
        n: проверяемое число
        k: количество раундов теста (чем больше, тем точнее)
        Возвращает True, если число вероятно простое, иначе False
        """
        if n < 2:
            return False
        if n in (2, 3):
            return True
        if n % 2 == 0:
            return False
        
        s, d = 0, n - 1
        while d % 2 == 0:
            s += 1
            d //= 2
        
        for _ in range(k):
            a = random.randint(2, n - 2)
            x = pow(a, d, n)
            if x in (1, n - 1):
                continue
            for _ in range(s - 1):
                x = pow(x, 2, n)
                if x == n - 1:
                    break
            else:
                return False
        return True
    
    def generate_prime(self) -> int:
        """
        Генерирует случайное простое число заданной длины
        Возвращает простое число
        """
        while True:
            p = random.getrandbits(self.key_length // 2)
            p |= (1 << (self.key_length // 2 - 1)) | 1
            if self.is_prime(p):
                return p
    
    def generate_keys(self) -> Tuple[Tuple[int, int], Tuple[int, int]]:
        """
        Генерирует пару ключей (публичный и приватный) для криптосистемы Пэйе
        Возвращает кортеж (public_key, private_key)
        public_key: (n, g) - модуль и параметр
        private_key: (lambda_val, mu) - секретные параметры для дешифрования
        """
        p = self.generate_prime()
        q = self.generate_prime()
        
        while p == q:
            q = self.generate_prime()
        
        n = p * q
        lambda_val = self.lcm(p - 1, q - 1)
        
        g = n + 1
        
        n_sq = n * n
        L_val = (pow(g, lambda_val, n_sq) - 1) // n
        mu = self.mod_inverse(L_val, n)
        
        public_key = (n, g)
        private_key = (lambda_val, mu)
        
        return public_key, private_key
    
    def L(self, x: int, n: int) -> int:
        """
        Функция L, используемая в дешифровании: L(x) = (x - 1) / n
        x: число, для которого вычисляется функция
        n: модуль
        Возвращает (x - 1) // n
        """
        return (x - 1) // n
    
    def encrypt(self, m: int, public_key: Tuple[int, int]) -> int:
        """
        Шифрует сообщение с использованием публичного ключа
        m: сообщение для шифрования (целое число)
        public_key: публичный ключ (n, g)
        Возвращает зашифрованное сообщение c
        """
        n, g = public_key
        n_sq = n * n
        
        while True:
            r = random.randint(1, n - 1)
            if self.gcd(r, n) == 1:
                break
        
        c = (pow(g, m, n_sq) * pow(r, n, n_sq)) % n_sq
        return c
    
    def decrypt(self, c: int, public_key: Tuple[int, int], private_key: Tuple[int, int]) -> int:
        """
        Расшифровывает сообщение с использованием приватного ключа
        c: зашифрованное сообщение
        public_key: публичный ключ (n, g)
        private_key: приватный ключ (lambda_val, mu)
        Возвращает расшифрованное сообщение m
        """
        n, g = public_key
        lambda_val, mu = private_key
        n_sq = n * n
        
        m = (self.L(pow(c, lambda_val, n_sq), n) * mu) % n
        return m
    
    def text_to_numbers(self, text: str) -> List[int]:
        """
        Преобразует текст в список чисел (коды ASCII)
        text: исходный текст
        Возвращает список целых чисел - ASCII коды символов
        """
        return [ord(char) for char in text]
    
    def numbers_to_text(self, numbers: List[int]) -> str:
        """
        Преобразует список чисел обратно в текст
        numbers: список ASCII кодов
        Возвращает строку из соответствующих символов
        """
        return ''.join(chr(num) for num in numbers)



```

Генерация ключей:
Выбираем два больших простых числа p и q

Вычисляем n = p * q (публичный модуль)

Вычисляем λ = НОК(p-1, q-1) (секретный параметр)

Выбираем g = n + 1 (публичный параметр)

Публичный ключ: (n, g)
Приватный ключ: (λ, μ)

Шифрование:
Для шифрования числа m:

Выбираем случайное r (1 < r < n)

Вычисляем: c = (gᵐ * rⁿ) mod n²

Важно: Одно и то же m с разными r дает разные c!

Дешифрование:
m = L(c^λ mod n²) * μ mod n
